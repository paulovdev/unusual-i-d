  initial: { y: "100%" },
  animate: (custom) => ({
    y: "0%",
    transition: {
      duration: 0.8,
      ease: [0.33, 1, 0.68, 1],
      delay: custom,
    },
  }),
};

const mediaReveal = {
  initial: {
    clipPath: "inset(0% 100% 0% 0%)",
  },
  animate: {
    clipPath: "inset(0% 0% 0% 0%)",
    transition: {
      duration: 0.9,
      ease: [0.76, 0, 0.24, 1],
    },
  },
  exit: {
    clipPath: "inset(0% 0% 0% 100%)",

    transition: {
      duration: 0.9,
      ease: [0.76, 0, 0.24, 1],
    },
  },
};
const Card = ({ work, index }) => {
  const righted = work.align === "right-0";
  const [card, setCard] = useState(true);
  const [activeMediaIndex, setActiveMediaIndex] = useState(0);
  const sectionRef = useRef(null);

  const menuAnim = {
    initial: {
      clipPath: righted ? "inset(0% 0% 0% 100%)" : "inset(0% 100% 0% 0%)",
      transition: {
        duration: 0.8,
        ease: [0.76, 0, 0.24, 1],
      },
    },
    animate: {
      clipPath: "inset(0% 0% 0% 0%)",
      transition: {
        duration: 0.8,
        ease: [0.76, 0, 0.24, 1],
      },
    },
  };

  const { ref, inView } = useInView({
    threshold: 0.9,
    triggerOnce: false,
  });

  useScroll({
    target: sectionRef,
    offset: ["start end", "end start"],
  });

  useEffect(() => {
    if (inView) {
      setCard(true);
    }
  }, [inView]);

  const mediaItems = useMemo(() => {
    const images = Array.isArray(work.image)
      ? work.image
      : work.image
        ? [work.image]
        : [];
    const videos = Array.isArray(work.videos)
      ? work.videos
      : work.videos
        ? [work.videos]
        : [];

    return [
      ...images.map((src) => ({
        type: "image",
        src,
      })),
      ...videos.map((src) => ({
        type: "video",
        src,
      })),
    ];
  }, [work.image, work.videos]);

  const hasMultipleMedia = mediaItems.length > 1;
  const activeMedia = mediaItems[activeMediaIndex] || null;

  const handlePrevMedia = () => {
    if (!mediaItems.length) return;

    setActiveMediaIndex((prev) =>
      prev === 0 ? mediaItems.length - 1 : prev - 1,
    );
  };

  const handleNextMedia = () => {
    if (!mediaItems.length) return;

    setActiveMediaIndex((prev) =>
      prev === mediaItems.length - 1 ? 0 : prev + 1,
    );
  };

  return (
    <div
      ref={sectionRef}
      className="relative h-[160vh]"
      style={{ zIndex: index + 1 }}
    >
      <div className="sticky top-0 h-screen w-full overflow-hidden">
        <div className="absolute inset-0 size-full overflow-hidden" ref={ref}>
          <AnimatePresence mode="sync">
            <motion.div
              key={`${activeMedia?.type}-${activeMediaIndex}`}
              variants={mediaReveal}
              initial="initial"
              animate="animate"
              exit="exit"
              className="absolute inset-0 size-full"
            >
              {activeMedia?.type === "video" ? (
                <video
                  src={activeMedia.src}
                  autoPlay
                  muted
                  loop
                  playsInline
                  preload="metadata"
                  className="size-full object-cover brightness-75"
                />
              ) : activeMedia?.type === "image" ? (
                <Image
                  src={activeMedia.src}
                  fill
                  alt={work.alt}
                  className="size-full object-cover brightness-75"
                  sizes="100vw"
                  priority={index === 0}
                />
              ) : null}
            </motion.div>
          </AnimatePresence>
        </div>

        <motion.div
          whileTap={{ scale: 1.1 }}
          transition={{
            duration: 0.1,
            ease: [0.76, 0, 0.24, 1],
          }}
          className={`absolute top-0 left-1/2 -translate-x-1/2 z-20 max-md:p-2
          ${
            righted
              ? "max-md:left-[calc(0%+32px)]"
              : "max-md:left-[calc(100%-32px)]"
          } group`}
        >
          <motion.button
            onClick={() => setCard(!card)}
            className="p-3 bg-p/25 backdrop-blur-2xl rounded-sm cursor-default"
            initial={{ scale: 0 }}
            animate={{
              scale: inView ? 1 : 0,
            }}
            transition={{
              duration: 0.8,
              ease: [0.76, 0, 0.24, 1],
            }}
          >
            <motion.span
              animate={{
                rotate: card ? 0 : 180,
              }}
              transition={{ duration: 0.8, ease: [0.76, 0, 0.24, 1] }}
              className="flex"
            >
              <RiFullscreenExitLine className="text-s text-[24px]" />
            </motion.span>
          </motion.button>
        </motion.div>

        <motion.div
          className={`absolute p-15 ${work.align} w-[50vw] h-full backdrop-blur-2xl z-10
          max-md:p-5 max-md:w-[85vw]`}
          variants={menuAnim}
          initial="initial"
          animate={inView && card ? "animate" : "initial"}
        >
          <div className="size-full flex flex-col items-end justify-between max-md:justify-start max-md:gap-15">
            <div className="mb-113 w-full flex flex-col gap-15 max-md:mb-10">
              <div className="flex items-center">
                <span
                  className="font-azeret font-medium 
                  text-s text-[14px] tracking-[0.01em] leading-[1.1] uppercase truncate
                  "
                >
                  {work.category}
                </span>
                <span className="mx-5 text-s">/</span>
                <span
                  className="font-azeret font-medium 
                  text-s text-[14px] tracking-[0.01em] leading-[1.1] uppercase
                  "
                >
                  2024
                </span>
              </div>

              <h2
                className="font-i-sans font-normal 
                text-s text-[72px] tracking-[-0.07em] leading-none
                max-md:text-[42px]"
              >
                {work.title}
                <span className="relative -top-10 left-2 text-[18px] max-md:-top-6">
                  {work.mark}
                </span>
              </h2>
            </div>

            <div className="w-full flex items-start max-md:flex-col max-md:gap-5">
              <div className="flex-1">
                <div className="size-fit flex items-center gap-2">
                  <span className="size-2 bg-s rounded-[1px]" />
                  <p
                    className="font-azeret font-medium 
                    text-s text-[14px] tracking-[0.05em] leading-none uppercase
                    "
                  >
                    about work
                  </p>
                </div>
              </div>

              <div className="flex flex-col gap-15">
                <TextAnimated
                  phrases={work.description}
                  variants={textSlide}
                  animate={inView && card}
                  as="span"
                  className="flex flex-col"
                  lineClassName="max-w-150 mb-5 font-azeret font-medium 
                  text-s text-[14px] tracking-[0.05em] leading-[1.2] uppercase 
                  "
                  wordClassName="mr-2"
                  wordDelay={0.035}
                  lineDelay={0.035}
                />

                <div className="flex items-center gap-2">
                  <motion.button
                    whileTap={{ scale: 1.1 }}
                    className="size-12 bg-s rounded-sm flex items-center justify-center"
                    onClick={handlePrevMedia}
                    type="button"
                    aria-label="Previous media"
                    disabled={!hasMultipleMedia}
                  >
                    <RiArrowLeftLongLine className="text-[20px]" />
                  </motion.button>

                  <motion.button
                    whileTap={{ scale: 1.1 }}
                    className="size-12 bg-s rounded-sm flex items-center justify-center"
                    onClick={handleNextMedia}
                    type="button"
                    aria-label="Next media"
                    disabled={!hasMultipleMedia}
                  >
                    <RiArrowRightLongLine className="text-[20px]" />
                  </motion.button>

                  <Button
                    text="visit"
                    bg="bg-p"
                    textColor="text-s"
                    iconColor="text-s"
                    hoverBg="bg-s"
                    hoverTextColor="text-p"
                    hoverIconColor="text-p"
                  />
                </div>
              </div>
            </div>
          </div>
        </motion.div>
      </div>
    </div>
  );
};

const HomeWorks = () => {
  const { ref, inView } = useInView({
    threshold: 0.85,
    triggerOnce: false,
  });

  return (
    <section className="relative">
      <motion.div
        ref={ref}
        className="absolute inset-0 w-full h-screen flex items-start justify-center gap-2 z-10 pointer-events-none"
        initial={{ opacity: 0, y: -10 }}
        animate={{
          opacity: inView ? 0 : 1,
          y: inView ? 0 : -10,
          transition: { duration: 0.25, delay: 0.25 },
        }}
      >
        <div className="relative top-10 flex items-center gap-2">
          <span className="size-2 bg-s rounded-[1px]" />
          <p className="max-w-125 font-azeret font-medium text-s text-[14px] tracking-[0.05em] leading-none uppercase">
            Selected works
          </p>
        </div>
      </motion.div>

      {worksData.map((work, i) => (
        <Card key={work.id} work={work} index={i} />
      ))}
    </section>
  );
};

export default HomeWorks;
